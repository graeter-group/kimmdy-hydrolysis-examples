top = Topology.from_path(

bonds['bond_type'] = bonds.apply(
  lambda row: "-".join(([row['atom_i'], row['atom_j']])),
  axis=1,
)
bonds['residue_pair'] = bonds.apply(
  lambda row: "-".join(([top.atoms[str(row['ai'])].residue, top.atoms[str(row['aj'])].residue])),
  axis=1,
)
bonds['pro_first'] = bonds.apply(
  lambda row: top.atoms[str(row['ai'])].residue in ['PRO', 'HYP'],
  axis=1,
)
bonds['pro_second'] = bonds.apply(
  lambda row: top.atoms[str(row['aj'])].residue in ['PRO', 'HYP'],
  axis=1,
)
bonds['pro_position'] = bonds.apply(
  lambda row: 'both' if row['pro_first'] and row['pro_second'] else ('C' if row['pro_first'] else ('N' if row['pro_second'] else 'none')),

)
